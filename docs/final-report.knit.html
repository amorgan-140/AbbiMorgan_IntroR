Coffee Trends in the US

Introduction

As busy and tired scientist, many of us rely on coffee as a lifeline to get us through those long experiment or writing days. A data set that surveyed ~4,000 individuals, looked at how different demographics (such as age, gender, political affiliation, ect.) feel about different aspects of coffee (like additives, dairy preference, brew method, ect.). I wanted to use this final as a way to practice organizing the data, using different plotting methods, and some stats. Enjoy!

Organize the data

data_path <- "data"
csv_files <- list.files(path = data_path, pattern = "\\.csv$", full.names = TRUE)

for (file in csv_files) {
  df_name <- tools::file_path_sans_ext(basename(file))
  
  df <- read_csv(file) %>%
    rename_with(~ str_replace_all(.x, "_", " "))
  
  assign(df_name, df, envir = .GlobalEnv)
}

count_by_category <- function(df, column) {
  df %>%
    drop_na({{ column }}) %>%
    group_by({{ column }}) %>%
    summarize(n = n()) %>%
    arrange(desc(n))
}

# Summarize checkbox-style logical columns (TRUE/FALSE)
summarize_checkboxes <- function(df, pattern) {
  df %>%
    summarize(across(matches(pattern), ~ sum(.x, na.rm = TRUE))) %>%
    pivot_longer(everything(), names_to = "option", values_to = "count") %>%
    arrange(desc(count))
}

# Convert checkbox columns into long format for plotting (ignore NA)
checkbox_long <- function(df, pattern) {
  df %>%
    pivot_longer(
      cols = matches(pattern),
      names_to = "option",
      values_to = "selected"
    ) %>%
    filter(selected == TRUE) %>%
    count(option, sort = TRUE)
}


# ANALYSIS: Gender × Brew Method

brew_by_gender <- GACTT_RESULTS_ANONYMIZED_v2 %>%
  pivot_longer(
    cols = matches("^How do you brew coffee at home\\? \\("),
    names_to = "brew_method",
    values_to = "selected"
  ) %>%
  filter(selected == TRUE) %>%      # keep TRUE selections only
  drop_na(Gender) %>%               # remove NA gender
  count(Gender, brew_method, sort = TRUE)


# Clean brew method labels (extract text inside parentheses)
brew_by_gender <- brew_by_gender %>%
  mutate(
    brew_method = str_extract(brew_method, "\\((.*)\\)") %>%
      str_remove_all("[()]")
  )

# Combine "Other" + "Prefer not to say"
GACTT_RESULTS_ANONYMIZED_v2 <- GACTT_RESULTS_ANONYMIZED_v2 %>%
  mutate(
    Gender = case_when(
      Gender %in% c("Other (please specify)", "Prefer not to say") ~ "Other / Prefer not to say",
      TRUE ~ Gender
    )
  )

# PLOT

p <- brew_by_gender %>%
  ggplot(aes(x = brew_method, y = n, fill = Gender)) +
  geom_col(position = "dodge") +
  theme_minimal() +
  labs(
    title = "Brew method vs. Gender",
    x = "Brew Method",
    y = "Count"
  ) +
  theme(axis.text.x = element_text(angle = 45, hjust = 1))

print(p)

brew_donut <- brew_by_gender %>%
  group_by(Gender) %>%
  mutate(
    percent = n / sum(n) * 100
  )

ggplot(brew_donut, aes(x = 2, y = percent, fill = brew_method)) +
  geom_col(color = "white") +
  coord_polar(theta = "y") +
  facet_wrap(~ Gender) +
  xlim(0.5, 2.5) +   # creates the donut hole
  theme_void() +
  labs(
    title = "How People Brew Coffee at Home, by Gender",
    fill = "Brew Method"
  ) +
  theme(
    strip.text = element_text(size = 14, face = "bold"),
    plot.title = element_text(size = 16, face = "bold", hjust = 0.5)
  )

# ranks the categories
brew_ranked <- brew_by_gender %>%
  group_by(Gender) %>%
  arrange(Gender, desc(n)) %>%
  mutate(
    rank = row_number(),
    percent = round(n / sum(n) * 100, 1)
  ) %>%
  select(rank, Gender, brew_method, n, percent)
